2. Implementación en Backend (Express & Node)

Componentes desacoplados según el Principio de Responsabilidad Única (SRP):

// 1. Interceptor de Verificación JWT (verifyToken)
export const verifyToken = (req, res, next) => {
  const token = req.headers.authorization?.split(' ')[1];
  if (!token) return res.status(401).json({ message: 'No token' });
  req.userRole = jwt.verify(token, process.env.JWT_SECRET).rol_id;
  next();
};

// 2. Interceptor de Control de Rol (requireRole)
export const requireRole = (roleId) => (req, res, next) => {
  if (req.userRole !== roleId) return res.status(403).json({ error: 'Acceso denegado' });
  next();
};

// 3. Declaración en Router
router.post('/admin/config', verifyToken, requireRole(1), adminController.save);

3. Implementación en Frontend (React Router)

Envoltorio reutilizable para control de acceso en la navegación:

// ProtectedRoute.tsx
export const ProtectedRoute = ({ children, requiredRole }) => {
  const { isAuthenticated, user, loading } = useAuth();
  if (loading) return <div>Cargando...</div>;
  if (!isAuthenticated) return <Navigate to="/login" replace />;
  if (requiredRole && user?.rol !== requiredRole) return <Navigate to="/unauthorized" replace />;
  return <>{children}</>;
};

// Declaración en Router
<Routes>
  <Route path="/admin" element={
    <ProtectedRoute requiredRole={1}>
      <AdminPage />
    </ProtectedRoute>
  } />
</Routes>
